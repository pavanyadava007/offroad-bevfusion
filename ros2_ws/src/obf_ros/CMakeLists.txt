cmake_minimum_required(VERSION 3.18)
project(obf_ros LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(vision_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(CUDAToolkit REQUIRED)
find_path(TENSORRT_INCLUDE NvInfer.h HINTS ${TENSORRT_ROOT}/include /usr/include/x86_64-linux-gnu /usr/include)
find_library(TENSORRT_LIB nvinfer HINTS ${TENSORRT_ROOT}/lib /usr/lib/x86_64-linux-gnu)
find_library(TENSORRT_PLUGIN_LIB nvinfer_plugin HINTS ${TENSORRT_ROOT}/lib /usr/lib/x86_64-linux-gnu)
set(OBF_CPP ${CMAKE_CURRENT_SOURCE_DIR}/../../../cpp)
add_library(obf_trt SHARED ${OBF_CPP}/src/trt_runner.cpp)
target_include_directories(obf_trt PUBLIC ${OBF_CPP}/include ${TENSORRT_INCLUDE})
target_link_libraries(obf_trt PUBLIC ${TENSORRT_LIB} ${TENSORRT_PLUGIN_LIB} CUDA::cudart)
add_executable(obf_node src/obf_node.cpp)
target_link_libraries(obf_node obf_trt)
set_target_properties(obf_node PROPERTIES INSTALL_RPATH "\$ORIGIN")  # obf_trt installs beside the binary
ament_target_dependencies(obf_node rclcpp vision_msgs nav_msgs sensor_msgs visualization_msgs geometry_msgs)
install(TARGETS obf_node obf_trt DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/replay_sensors.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch rviz DESTINATION share/${PROJECT_NAME})
ament_package()
