cmake_minimum_required(VERSION 3.18)
project(obf_trt_runner LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CUDAToolkit REQUIRED)
find_package(Threads REQUIRED)
# TensorRT: set -DTENSORRT_ROOT=/usr/lib/x86_64-linux-gnu or the tarball dir
find_path(TENSORRT_INCLUDE NvInfer.h HINTS ${TENSORRT_ROOT}/include /usr/include/x86_64-linux-gnu /usr/include)
find_library(TENSORRT_LIB nvinfer HINTS ${TENSORRT_ROOT}/lib /usr/lib/x86_64-linux-gnu)
find_library(TENSORRT_PLUGIN_LIB nvinfer_plugin HINTS ${TENSORRT_ROOT}/lib /usr/lib/x86_64-linux-gnu)
if(NOT TENSORRT_INCLUDE OR NOT TENSORRT_LIB)
  message(FATAL_ERROR "TensorRT not found; pass -DTENSORRT_ROOT=...")
endif()
add_library(obf_trt SHARED src/trt_runner.cpp)
target_include_directories(obf_trt PUBLIC include ${TENSORRT_INCLUDE})
target_link_libraries(obf_trt PUBLIC ${TENSORRT_LIB} ${TENSORRT_PLUGIN_LIB} CUDA::cudart)
add_executable(obf_runner src/main.cpp)
target_link_libraries(obf_runner PRIVATE obf_trt)
install(TARGETS obf_trt obf_runner LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)

# Real-time latency/jitter benchmark under CPU/memory/GPU interference (see results/REALTIME.md).
option(OBF_BUILD_BENCH "build latency_bench, cpu_hog, gpu_hog and the rt_stats ctest" ON)
if(OBF_BUILD_BENCH)
enable_testing()
add_library(obf_rt_stats STATIC bench/rt_stats.cpp)
target_include_directories(obf_rt_stats PUBLIC bench)
add_executable(test_rt_stats bench/test_rt_stats.cpp)
target_link_libraries(test_rt_stats PRIVATE obf_rt_stats)
add_test(NAME rt_stats COMMAND test_rt_stats)
add_library(obf_hog_kernel STATIC bench/hog_kernel.cu)
target_include_directories(obf_hog_kernel PUBLIC bench)
set_target_properties(obf_hog_kernel PROPERTIES CUDA_ARCHITECTURES "75;86;87;89")
add_executable(latency_bench bench/latency_bench.cpp)
target_link_libraries(latency_bench PRIVATE obf_trt obf_rt_stats obf_hog_kernel CUDA::nvml Threads::Threads)
add_executable(gpu_hog bench/gpu_hog.cpp)
target_link_libraries(gpu_hog PRIVATE obf_hog_kernel CUDA::cudart)
add_executable(cpu_hog bench/cpu_hog.cpp)
target_link_libraries(cpu_hog PRIVATE Threads::Threads)
endif()
