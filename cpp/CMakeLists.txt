cmake_minimum_required(VERSION 3.18)
project(obf_trt_runner LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CUDAToolkit REQUIRED)
# TensorRT: set -DTENSORRT_ROOT=/usr/lib/x86_64-linux-gnu or the tarball dir
find_path(TENSORRT_INCLUDE NvInfer.h HINTS ${TENSORRT_ROOT}/include /usr/include/x86_64-linux-gnu /usr/include)
find_library(TENSORRT_LIB nvinfer HINTS ${TENSORRT_ROOT}/lib /usr/lib/x86_64-linux-gnu)
find_library(TENSORRT_PLUGIN_LIB nvinfer_plugin HINTS ${TENSORRT_ROOT}/lib /usr/lib/x86_64-linux-gnu)
if(NOT TENSORRT_INCLUDE OR NOT TENSORRT_LIB)
  message(FATAL_ERROR "TensorRT not found; pass -DTENSORRT_ROOT=...")
endif()
add_library(obf_trt SHARED src/trt_runner.cpp)
target_include_directories(obf_trt PUBLIC include ${TENSORRT_INCLUDE})
target_link_libraries(obf_trt PUBLIC ${TENSORRT_LIB} ${TENSORRT_PLUGIN_LIB} CUDA::cudart)
add_executable(obf_runner src/main.cpp)
target_link_libraries(obf_runner PRIVATE obf_trt)
install(TARGETS obf_trt obf_runner LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
